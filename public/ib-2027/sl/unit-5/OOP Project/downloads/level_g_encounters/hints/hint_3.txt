CHOOSE_ACTION:

def choose_action(self, options):
    print("\nWhat do you do?")
    for i, option in enumerate(options):
        print(f"  {i+1}. {option}")
    
    while True:
        try:
            choice = int(input("Choice: ")) - 1
            if 0 <= choice < len(options):
                return options[choice]
            else:
                print("Invalid choice.")
        except ValueError:
            print("Enter a number.")

COMPUTE_DAMAGE:

def compute_damage(self, base_damage, armour):
    damage = base_damage - armour
    return max(0, damage)  # Never negative!

---------------------------------------------
KEY CONCEPTS
---------------------------------------------

✓ max(0, value) ensures never below 0
✓ Return the actual option string, not the index
✓ Input validation with try/except
✓ Combat logic stays simple - engine handles the rest
