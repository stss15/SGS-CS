{\color{sgsGrey}Checklist worksheet. Write answers in your exercise book.\par}
\vspace{0.2em}
\hrule
\vspace{0.6em}
\NameLine

\vspace{0.6em}
\begin{StepBox}{Step 1: Open the dossier}
\begin{itemize}[leftmargin=*,label=$\square$,nosep]
  \item Open: \url{https://sgs-science.web.app/ks3/year7/unit2/john-doe.html}
  \item Scan all 15 evidence images.
  \item For each image ask: \textit{What does this reveal?}
\end{itemize}
\end{StepBox}

\begin{StepBox}{Step 2: Set up your investigation table (in your book)}
Draw a table with these headings (one row per image):
\begin{itemize}[leftmargin=*,label=$\square$,nosep]
  \item Image \#
  \item What does it reveal?
  \item Public or private? Why?
\end{itemize}
\end{StepBox}

\begin{StepBox}{Step 3: Find 10+ clues}
Your list must include:
\begin{itemize}[leftmargin=*,label=$\square$,nosep]
  \item John's name
  \item John's date of birth (day + month)
  \item The city John lives in
\end{itemize}
Then add any other clues you can work out (hobbies, places, routine, family info, etc.).
\end{StepBox}

\begin{StepBox}{Step 4: Decide public vs private}
For each clue, explain whether it should be public or private:
\begin{itemize}[leftmargin=*,label=$\square$,nosep]
  \item Could it help someone identify or locate John?
  \item Could it harm his safety or reputation?
  \item Is it \textit{nice to share} or \textit{risky to share}?
\end{itemize}
\end{StepBox}

\begin{StepBox}{Step 5: Reflection (answer in your book)}
\textbf{A.} What picture can you build of John? (4--5 bullet points)\par
\textbf{B.} Choose 3 things that should have stayed private. For each: one sentence explaining the risk.
\end{StepBox}

\begin{StepBox}{Step 6: Partner discussion}
Share one surprising clue you found. Compare what you thought should stay private.
\end{StepBox}

\begin{InfoBox}{Learning check}
\begin{itemize}[leftmargin=*,label=$\square$,nosep]
  \item I can explain the difference between public and private information online.
  \item I can explain how small clues build a picture of someone.
  \item I can explain why sharing too much can be risky.
\end{itemize}
\end{InfoBox}
