\begin{SGSCallout}{What you are learning}
People behave differently on different platforms. This activity helps you judge online behaviour, explain your reasoning, and understand when anonymity is helpful.
\end{SGSCallout}

\begin{SGSStep}{1}{Understand the Swipes}
Each card has a \textbf{scenario} (the situation) and an \textbf{action} (what someone did). Read both sides, then swipe based on your judgement:

\vspace{0.3em}
\renewcommand{\arraystretch}{1.2}
\begin{tabularx}{\textwidth}{|p{2.2cm}|X|p{3.6cm}|}
\hline
\textbf{Swipe} & \textbf{Category} & \textbf{Meaning} \\
\hline
$\leftarrow$ & Good \& Use Your Name & No need to hide who you are \\ \hline
$\rightarrow$ & Good but Stay Anonymous & Better to use a nickname \\ \hline
$\uparrow$ & It Depends... & This one is complicated \\ \hline
$\downarrow$ & Not OK Online & This should not happen \\ \hline
\end{tabularx}
\end{SGSStep}

\begin{SGSStep}{2}{Play Swipe Scenarios}
\textbf{Open the activity:} \url{https://sgs-science.web.app/ks3/year7/unit2/Online-behaviour.html}

\begin{itemize}[leftmargin=*,label=$\square$]
  \item Read the scenario and action carefully.
  \item Swipe in the direction that matches your judgement.
  \item If you are unsure, use \textbf{It Depends...} and explain why later.
\end{itemize}
\end{SGSStep}

\begin{SGSStep}{3}{Exercise Book Tasks}
After completing the game, answer the prompts below:

\begin{itemize}[leftmargin=*,label=$\square$]
  \item \textbf{From ``Not OK Online'':} Choose one card and explain why this behaviour is never acceptable.
  \item \textbf{From ``Good \& Use Your Name'':} Choose one card and explain why being yourself is good in this situation.
  \item \textbf{From ``Good but Stay Anonymous'':} Choose one card and explain why staying anonymous makes sense.
  \item \textbf{From ``It Depends...'':} Choose 3--4 cards and explain what makes each one complicated.
\end{itemize}
\end{SGSStep}

\begin{itemize}[leftmargin=*,label=$\square$]
