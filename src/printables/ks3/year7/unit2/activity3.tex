\vspace{0.2em}
\hrule
\vspace{0.6em}
\NameLine

\vspace{0.6em}
\begin{StepBox}{Step 1: Launch the activity}
\begin{itemize}[leftmargin=*,label=$\square$,nosep]
  \item Open: \url{https://sgs-science.web.app/ks3/year7/unit2/Online-behaviour.html}
  \item Read each card: scenario + action.
  \item Swipe to sort the card.
\end{itemize}
\end{StepBox}

\begin{StepBox}{Step 2: What the swipes mean}
\begin{itemize}[leftmargin=*,label=$\square$,nosep]
  \item $\leftarrow$ Good \& use your name (no need to hide who you are)
  \item $\rightarrow$ Good but stay anonymous (use a nickname)
  \item $\uparrow$ It depends (complicated)
  \item $\downarrow$ Not OK online (should not happen)
\end{itemize}
\end{StepBox}

\begin{StepBox}{Step 3: Exercise book tasks}
\begin{itemize}[leftmargin=*,label=$\square$,nosep]
  \item From ``Not OK online'': choose 1 card and explain why it is never acceptable.
  \item From ``Good \& use your name'': choose 1 card and explain why it is OK to be yourself.
  \item From ``Good but stay anonymous'': choose 1 card and explain why anonymity makes sense.
  \item From ``It depends'': choose 3--4 cards and explain what makes each one complicated.
\end{itemize}
\end{StepBox}

\begin{StepBox}{Step 4: Reflection (answer in your book)}
\textbf{A.} Why do people behave differently on different platforms?\par
\textbf{B.} What is ``context collapse''? Give an example.
\end{StepBox}

\begin{InfoBox}{Learning check}
\begin{itemize}[leftmargin=*,label=$\square$,nosep]
  \item I can explain when anonymity is helpful.
  \item I can explain why behaviour changes across apps.
  \item I can explain what context collapse means.
\end{itemize}
\end{InfoBox}
