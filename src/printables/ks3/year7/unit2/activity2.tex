\begin{StepBox}{Step 1: Launch MiniPhone OS}
\begin{itemize}[leftmargin=*,label=$\square$,nosep]
  \item Open: \url{https://sgs-science.web.app/ks3/year7/unit2/activity2/index.html}
  \item Open \textbf{Messages} and follow the missions.
  \item On a laptop, use the \textbf{Notes} panel (right). On mobile, tap \textbf{Notes}.
  \item Use \textbf{Case File} to review what data was collected.
\end{itemize}
\end{StepBox}

\begin{StepBox}{Step 2: Set up your investigation log (in your book)}
Draw a table with these headings (one row per app):
\begin{itemize}[leftmargin=*,label=$\square$,nosep]
  \item App
  \item What data did it ask for?
  \item Why might it want it?
  \item What happens if you say NO?
\end{itemize}
\end{StepBox}

\begin{StepBox}{Step 3: Missions (what to look for)}
\begin{itemize}[leftmargin=*,label=$\square$,nosep]
  \item \textbf{Browser:} Try both cookie choices. Tap a few items. Does \textbf{Sponsored} change? Use the cookie button to switch.
  \item \textbf{Connectify:} Tap highlighted clauses. Decide: agree or decline. If you agree: explore tabs and look for \textbf{Sponsored}.
  \item \textbf{MapMe:} Try \textit{Precise} vs \textit{Approximate}. Tap pins. Compare dot vs big circle. Use the crosshair button to switch.
  \item \textbf{Super Puzzle Quest:} Notice tracking/camera/contacts prompts. Watch one ad for coins. What changes when you say NO?
\end{itemize}
\end{StepBox}

\begin{StepBox}{Step 4: Reflection (answer in your book)}
\textbf{A.} Which app was the ``greediest''? Was it justified?\par
\textbf{B.} One privacy vs convenience trade-off you noticed.
\end{StepBox}

\begin{StepBox}{Step 5: Partner discussion}
Compare your findings. Which permission choices felt most risky? Why?
\end{StepBox}

\begin{InfoBox}{Learning check}
\begin{itemize}[leftmargin=*,label=$\square$,nosep]
  \item I can describe what cookies and permissions do.
  \item I can explain why apps ask for data.
  \item I can explain privacy vs convenience.
\end{itemize}
\end{InfoBox}
